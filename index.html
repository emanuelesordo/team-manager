DIRECTORY SQUADRE UTILIZZABILE PRIMA DEL LOGIN -- Solo squadre scopribili e seguibili. -- --------------------------------------------------------- create or replace function public.registration_team_directory() returns table( team_id uuid, team_name text, team_short_name text, organization_name text, logo_url text, display_code text ) language sql stable security definer set search_path='' as $$ select t.id, t.name, t.short_name, o.name, case when t.inherit_organization_branding then coalesce(o.logo_url,t.logo_url) else coalesce(t.logo_url,o.logo_url) end, case when t.inherit_organization_branding then coalesce(o.display_code,t.short_name) else coalesce(t.short_name,o.display_code) end from public.teams t left join public.organizations o on o.id=t.organization_id where t.is_discoverable and t.allow_fan_follow order by coalesce(o.name,''),t.name; $$; revoke all on function public.registration_team_directory() from public; grant execute on function public.registration_team_directory() to anon, authenticated; -- --------------------------------------------------------- -- 2. RICHIESTE ACCESSO STAFF A SQUADRA ESISTENTE -- --------------------------------------------------------- create table if not exists public.team_staff_access_requests ( id uuid primary key default gen_random_uuid(), team_id uuid not null references public.teams(id) on delete cascade, profile_id uuid not null references public.profiles(id) on delete cascade, requested_role_code text not null check(requested_role_code in ('management','coach','assistant_coach')), message text, status text not null default 'pending' check(status in ('pending','approved','rejected','cancelled')), requested_at timestamptz not null default now(), reviewed_by uuid references public.profiles(id) on delete set null, reviewed_at timestamptz, review_reason text, created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); drop trigger if exists team_staff_access_requests_set_updated_at on public.team_staff_access_requests; create trigger team_staff_access_requests_set_updated_at before update on public.team_staff_access_requests for each row execute function public.set_updated_at(); create unique index if not exists team_staff_access_requests_one_pending_idx on public.team_staff_access_requests(team_id,profile_id) where status='pending'; create index if not exists team_staff_access_requests_team_status_idx on public.team_staff_access_requests(team_id,status,requested_at desc); alter table public.team_staff_access_requests enable row level security; revoke all on public.team_staff_access_requests from anon; revoke all on public.team_staff_access_requests from authenticated; grant select on public.team_staff_access_requests to authenticated; drop policy if exists team_staff_access_requests_read_029 on public.team_staff_access_requests; create policy team_staff_access_requests_read_029 on public.team_staff_access_requests for select to authenticated using( profile_id=(select auth.uid()) or private.user_has_permission( (select auth.uid()), team_id, 'staff.manage', current_date ) ); -- --------------------------------------------------------- -- 3. INVIO RICHIESTA STAFF -- L'utente deve già seguire la squadra come Fan. -- --------------------------------------------------------- create or replace function public.request_team_staff_access( p_team_id uuid, p_requested_role_code text, p_message text default null ) returns uuid language plpgsql security definer set search_path='' as $$ declare v_request_id uuid; v_member_id uuid; begin if (select auth.uid()) is null then raise exception 'authentication required'; end if; if p_requested_role_code not in ('management','coach','assistant_coach') then raise exception 'invalid requested staff role'; end if; select tm.id into v_member_id from public.team_members tm where tm.team_id=p_team_id and tm.profile_id=(select auth.uid()) and tm.is_active limit 1; if v_member_id is null then raise exception 'follow the team before requesting staff access'; end if; if exists( select 1 from public.staff_assignments sa where sa.team_id=p_team_id and sa.team_member_id=v_member_id and sa.status='active' and sa.role_code in ('management','coach','assistant_coach') and current_date>=sa.start_date and (sa.actual_end_date is null or current_date<=sa.actual_end_date) and (sa.planned_end_date is null or current_date<=sa.planned_end_date) ) then raise exception 'you already have an active staff role in this team'; end if; if exists( select 1 from public.team_staff_access_requests r where r.team_id=p_team_id and r.profile_id=(select auth.uid()) and r.status='pending' ) then raise exception 'you already have a pending staff-access request'; end if; insert into public.team_staff_access_requests( team_id, profile_id, requested_role_code, message ) values( p_team_id, (select auth.uid()), p_requested_role_code, nullif(trim(p_message),'') ) returning id into v_request_id; insert into public.audit_log( team_id,actor_id,action,entity_type,entity_id,after_data ) values( p_team_id, (select auth.uid()), 'team_staff_access_requested', 'team_staff_access_request', v_request_id::text, jsonb_build_object( 'requested_role_code',p_requested_role_code ) ); return v_request_id; end; $$; revoke all on function public.request_team_staff_access(uuid,text,text) from public; grant execute on function public.request_team_staff_access(uuid,text,text) to authenticated; create or replace function public.cancel_team_staff_access_request( p_request_id uuid ) returns void language plpgsql security definer set search_path='' as $$ begin update public.team_staff_access_requests set status='cancelled', reviewed_at=now() where id=p_request_id and profile_id=(select auth.uid()) and status='pending'; if not found then raise exception 'pending request not found'; end if; end; $$; revoke all on function public.cancel_team_staff_access_request(uuid) from public; grant execute on function public.cancel_team_staff_access_request(uuid) to authenticated; -- --------------------------------------------------------- -- 4. APPROVAZIONE STAFF -- Crea l'incarico che attiva i permessi del ruolo richiesto. -- --------------------------------------------------------- create or replace function public.review_team_staff_access_request( p_request_id uuid, p_action text, p_reason text default null ) returns void language plpgsql security definer set search_path='' as $$ declare v_request public.team_staff_access_requests%rowtype; v_member_id uuid; v_season public.seasons%rowtype; begin select * into v_request from public.team_staff_access_requests where id=p_request_id for update; if v_request.id is null then raise exception 'request not found'; end if; if v_request.status<>'pending' then raise exception 'request already reviewed'; end if; if not private.user_has_permission( (select auth.uid()), v_request.team_id, 'staff.manage', current_date ) then raise exception 'permission denied'; end if; if p_action not in ('approve','reject') then raise exception 'invalid review action'; end if; if p_action='reject' then update public.team_staff_access_requests set status='rejected', reviewed_by=(select auth.uid()), reviewed_at=now(), review_reason=nullif(trim(p_reason),'') where id=v_request.id; else select tm.id into v_member_id from public.team_members tm where tm.team_id=v_request.team_id and tm.profile_id=v_request.profile_id limit 1; if v_member_id is null then raise exception 'team membership not found'; end if; update public.team_members set is_active=true, is_hidden=false, left_at=null, updated_at=now() where id=v_member_id; select * into v_season from public.seasons s where s.team_id=v_request.team_id order by case when s.status='active' then 0 when current_date between s.start_date and s.end_date then 1 else 2 end, s.start_date desc limit 1; if v_season.id is null then raise exception 'no season available for this team'; end if; if not exists( select 1 from public.staff_assignments sa where sa.team_id=v_request.team_id and sa.team_member_id=v_member_id and sa.role_code=v_request.requested_role_code and sa.status='active' and (sa.actual_end_date is null or sa.actual_end_date>=current_date) and (sa.planned_end_date is null or sa.planned_end_date>=current_date) ) then insert into public.staff_assignments( team_id, season_id, team_member_id, role_code, start_date, planned_end_date, status, created_by ) values( v_request.team_id, v_season.id, v_member_id, v_request.requested_role_code, current_date, case when v_request.requested_role_code='management' then null else v_season.end_date end, 'active', (select auth.uid()) ); end if; update public.team_staff_access_requests set status='approved', reviewed_by=(select auth.uid()), reviewed_at=now(), review_reason=nullif(trim(p_reason),'') where id=v_request.id; end if; insert into public.audit_log( team_id,actor_id,action,entity_type,entity_id,after_data ) values( v_request.team_id, (select auth.uid()), 'team_staff_access_reviewed', 'team_staff_access_request', v_request.id::text, jsonb_build_object( 'action',p_action, 'requested_role_code',v_request.requested_role_code, 'profile_id',v_request.profile_id ) ); end; $$; revoke all on function public.review_team_staff_access_request(uuid,text,text) from public; grant execute on function public.review_team_staff_access_request(uuid,text,text) to authenticated;
